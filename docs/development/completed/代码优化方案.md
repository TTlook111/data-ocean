## 一、现状分析总结

### 1.1 前端代码问题

| 问题类型 | 具体表现 | 影响程度 |
|---------|---------|---------|
| **组件过大** | `QueryDatasourceView.vue` 超过 2300 行，职责过多 | 高 |
| **样式混合** | CSS 与逻辑代码混合，各页面断点不一致（1180px / 1100px / 600px） | 中 |
| **代码重复** | 多个页面有相似的加载状态、错误处理、分页逻辑 | 中 |
| **缺乏缓存策略** | API 响应无缓存，重复请求浪费带宽 | 低 |
| **动画过度** | GSAP 动画在低端设备可能卡顿，缺少性能检测 | 低 |

### 1.2 后端代码问题

| 问题类型 | 具体表现 | 影响程度 |
|---------|---------|---------|
| **服务类过大** | `KnowledgeDocServiceImpl` 779 行（God Class） | 高 |
| **重复模式** | 20+ 处 `requireXxx()` 存在性校验、`HashMap` 请求体构建 | 中 |
| **类型安全** | 大量 `Map<String, Object>` 返回值，缺乏类型检查 | 中 |
| **缓存问题** | `PermissionCalculatorImpl` 自实现缓存缺乏线程安全保障 | 中 |
| **异常处理** | 缺少统一的全局异常处理和错误码体系 | 中 |

### 1.3 Python 服务代码问题

| 问题类型 | 具体表现 | 影响程度 |
|---------|---------|---------|
| **模块组织** | 整体清晰，但 `graph.py`、`chunker.py` 等文件较长 | 低 |
| **配置管理** | 已集中化，但缺少运行时校验 | 低 |
| **错误处理** | 部分模块异常处理不完善，错误消息不够明确 | 中 |

### 1.4 跨层问题

| 问题类型 | 具体表现 | 影响程度 |
|---------|---------|---------|
| **安全防护** | 无 XSS/CSRF 防护策略、无依赖安全扫描 | 高 |
| **测试缺失** | 无前端测试、后端测试覆盖率不足、无 E2E 测试 | 高 |
| **可观测性** | 无 APM、无链路追踪、无结构化日志 | 中 |
| **回滚机制** | 无特性开关、无灰度发布策略 | 中 |

#### 2.3.2 日志脱敏

```python
# Python —— 日志脱敏处理器
import re

def mask_sensitive_data(event_dict):
    """脱敏敏感数据"""
    message = event_dict.get('event', '')
    # 手机号脱敏
    message = re.sub(r'1[3-9]\d{9}', lambda m: m.group()[:3] + '****' + m.group()[-4:], message)
    # 身份证脱敏
    message = re.sub(r'\d{17}[\dXx]', lambda m: m.group()[:6] + '********' + m.group()[-4:], message)
    event_dict['event'] = message
    return event_dict
```

public class DataOceanHealthIndicator implements HealthIndicator {
    @Autowired private DataSource dataSource;
    @Autowired private StringRedisTemplate redisTemplate;

    @Override
    public Health health() {
        // 检查数据库连接
        try (Connection conn = dataSource.getConnection()) {
            conn.isValid(1);

            .withDetail("redis", "connected")
            .build();
    }
}

#### 3.1.2 组件间通信方案

拆分后，组件间存在大量状态依赖。推荐使用 **Pinia Setup Store** 管理共享状态（更灵活，支持 Composables）：

**新增依赖**（`package.json`）：

```typescript
// stores/query.ts
import { ref, computed } from 'vue'
import { defineStore } from 'pinia'

export const useQueryStore = defineStore('query', () => {
  // --- State ---
  const selectedDatasourceId = ref<number | null>(null)
  const sessions = ref<LocalSession[]>([])
  const activeSessionId = ref<string | undefined>(undefined)
  const isQuerying = ref(false)
  const currentTaskId = ref<string | undefined>(undefined)

  // --- Getters ---
  const activeSession = computed(() =>
    sessions.value.find(s => s.id === activeSessionId.value)
  )
  const activeMessages = computed(() => activeSession.value?.messages || [])
  const latestResult = computed(() => {
    const msgs = activeMessages.value
    for (let i = msgs.length - 1; i >= 0; i--) {
      if (msgs[i].role === 'assistant' && msgs[i].queryResult) {
        return msgs[i].queryResult
      }
    }
    return null
  })

  // --- Actions ---
  function selectDatasource(id: number) { /* ... */ }
  function sendQuestion(question: string) { /* ... */ }
  function cancelQuery() { /* ... */ }

  return {
    selectedDatasourceId, sessions, activeSessionId, isQuerying, currentTaskId,
    activeSession, activeMessages, latestResult,
    selectDatasource, sendQuestion, cancelQuery,
  }
})
```

// composables/useApiRequest.ts
export function useApiRequest<T>(apiFn: (signal?: AbortSignal) => Promise<T>) {
  const data = ref<T | null>(null)
  const loading = ref(false)
  const error = ref<string | null>(null)
  let abortController: AbortController | null = null

  async function execute() {
    // 取消上一次未完成的请求（竞态处理）
    abortController?.abort()
    abortController = new AbortController()

    loading.value = true
    error.value = null
    try {
      data.value = await apiFn(abortController.signal)
    } catch (e: any) {
      if (e.name === 'AbortError') return // 主动取消，不报错
      error.value = extractError(e)
    } finally {
      loading.value = false
    }
  }

  function cancel() {
    abortController?.abort()
  }

  onBeforeUnmount(() => cancel())

  return { data, loading, error, execute, cancel }
}
```

const CACHE_DURATION = 5 * 60 * 1000 // 5 分钟
const cache = new Map<string, { data: any; timestamp: number }>()

/**
 * 带缓存的 API 请求
 * @param key 缓存键
 * @param apiFn API 请求函数
 * @param duration 缓存时长（毫秒）
 */
export async function fetchWithCache<T>(
  key: string,
  apiFn: () => Promise<T>,
  duration: number = CACHE_DURATION
): Promise<T> {
  const cached = cache.get(key)
  if (cached && Date.now() - cached.timestamp < duration) {
    return cached.data
  }

  const data = await apiFn()
  cache.set(key, { data, timestamp: Date.now() })
  return data
}

/** 清除指定缓存 */
export function clearCache(key: string): void {
  cache.delete(key)
}

/** 清除所有缓存 */
export function clearAllCache(): void {
  cache.clear()
}

// common/logging/LogActionAspect.java
@Aspect
@Component
@Slf4j
public class LogActionAspect {

    @Around("@annotation(logAction)")
    public Object around(ProceedingJoinPoint joinPoint, LogAction logAction) throws Throwable {
        String action = logAction.value();
        Object[] args = joinPoint.getArgs();

        if (logAction.logArgs()) {
            log.info("[{}] 开始 args={}", action, Arrays.toString(args));
        }

        long start = System.currentTimeMillis();
        try {
            Object result = joinPoint.proceed();
            long duration = System.currentTimeMillis() - start;

            if (logAction.logResult()) {
                log.info("[{}] 完成 durationMs={} result={}", action, duration, result);
            } else {
                log.info("[{}] 完成 durationMs={}", action, duration);
            }

            return result;
        } catch (Exception e) {
            long duration = System.currentTimeMillis() - start;
            log.error("[{}] 失败 durationMs={} error={}", action, duration, e.getMessage());
            throw e;
        }
    }
}

class SkillRegistry:
    """Skills 注册表"""
    _skills: Dict[str, Type[BaseSkill]] = {}

    @classmethod
    def register(cls, name: str):
        """注册 Skill 装饰器"""
        def decorator(skill_cls):
            cls._skills[name] = skill_cls
            return skill_cls
        return decorator

    @classmethod
    def get(cls, name: str) -> Type[BaseSkill]:
        if name not in cls._skills:
            raise ValueError(f"未知的 Skill: {name}")
        return cls._skills[name]

    @classmethod
    def list_all(cls) -> list[str]:
        return list(cls._skills.keys())
```
